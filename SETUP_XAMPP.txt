# ============================================================
# LUFLY Catalog - XAMPP Setup Instructions
# ============================================================
#
# إذا ظهرت لك "Internal Server Error"، اتبع الخطوات دي:
#
# 1. افتح ملف XAMPP:
#    xampp/apache/conf/httpd.conf
#
# 2. ابحث عن السطر:
#    <Directory "C:/xampp/htdocs">
#
# 3. تأكد إن AllowOverride مضبوط على All:
#    <Directory "C:/xampp/htdocs">
#        Options Indexes FollowSymLinks Includes ExecCGI
#        AllowOverride All        ← مهم جداً!
#        Require all granted
#    </Directory>
#
# 4. تأكد إن الموديلات دي مش كومنتيد (بدون # في أول السطر):
#    LoadModule rewrite_module modules/mod_rewrite.so
#    LoadModule php_module modules/php_module
#
# 5. احفظ httpd.conf وأعد تشغيل Apache من XAMPP Control Panel
#
# 6. افتح المتصفح على: http://localhost/lufly/
#
# ============================================================
# PHP Settings (في xampp/php/php.ini)
# ============================================================
#
# upload_max_filesize = 12M
# post_max_size = 128M
# max_file_uploads = 30
# extension=pdo_mysql
# extension=fileinfo
#
# بعد التعديل، أعد تشغيل Apache.
#
# ============================================================
# لو لسه المشكلة موجودة
# ============================================================
#
# جرب تفتح: http://localhost/lufly/index.php مباشرة
# لو فتح، المشكلة في .htaccess → راجع AllowOverride في httpd.conf
# لو مقفل، المشكلة في PHP → راجع error_log في xampp/apache/logs/
